React Native runs the same LiveThing client as the web — it has a global `WebSocket`, so `connect()` works out of the box, and the [React adapter](/recipes/framework-hooks/react) (`useStore`, `useIncoming`, requests) is exactly the same. There's no SSR to worry about either. What's different on mobile is the **environment**: an app gets backgrounded and the network flips constantly, so you'll want a [platform environment](/client/platform/overview) wired to React Native's signals. The socket below is Huddle's [client socket](/reference/huddle#the-client-socket) for React Native: it has the same primitives, with a mobile `platform()` environment instead of `WebEnvironment`.

## The React Native environment

A mobile environment maps `AppState` to visibility and NetInfo to network. Build it with `environment(...)`. Then pass it to [`platform()`](/client/platform/overview), alongside `reconnection()` and `keepalive()`:

::code-group

```ts [client/rn-environment.ts]
import { environment } from '@livething/client/environment'
import NetInfo from '@react-native-community/netinfo'
import { AppState } from 'react-native'

export const ReactNativeEnvironment = environment({
  onVisibilityChange(callback) {
    const sub = AppState.addEventListener('change', (s) =>
      callback(s === 'active'),
    )
    callback(AppState.currentState === 'active') // report the current state now
    return () => sub.remove()
  },
  onNetworkChange(callback) {
    return NetInfo.addEventListener((state) =>
      callback(state.isConnected ?? false),
    )
  },
})
```

```ts [client/socket.ts]
import {
  channels,
  client,
  clientRequests,
  connect,
  dedup,
  gaps,
  inbox,
  keepalive,
  platform,
  readReceipts,
  reconnection,
  replay,
  serverRequests,
  stream,
} from '@livething/client'
import { ReactNativeEnvironment } from './rn-environment'
import type { HuddleSocket } from '../server/huddle-definition'

export const socket = client<HuddleSocket>()(
  connect('wss://example.com/huddle'),
  channels(), // rooms and dms
  clientRequests(), // get-profile, delete-message
  serverRequests(), // confirm-delete
  stream(), // Scout's reply, below
  inbox(), // dm.receivers
  readReceipts(), // dm.readers
  dedup(),
  replay(),
  gaps(),
  keepalive(),
  platform({
    environment: ReactNativeEnvironment,
    backgroundDisconnectMs: 30_000, // drop the socket 30s after the app backgrounds
  }),
  reconnection(),
)
```

::

That gets you two mobile-critical behaviors for free:

- **Background disconnect** — when the app is backgrounded for a while, the socket closes (saving battery and a server connection) and reopens when the user returns. See [Background Disconnect](/client/platform/background-disconnect).
- **Network-aware reconnection** — on a flaky mobile network, retries pause while offline and fire the instant connectivity returns, instead of burning attempts. See [Network Aware](/client/platform/network-aware).

## Using the adapter

The React page's `useStore` works unchanged — create the socket once (above) and read any store through it:

```tsx [StatusDot.tsx]
import { StyleSheet, View } from 'react-native'
import { socket } from './socket'
import { useStore } from './use-store'

const styles = StyleSheet.create({
  green: { width: 8, height: 8, backgroundColor: 'green' },
  grey: { width: 8, height: 8, backgroundColor: 'grey' },
})

function StatusDot() {
  const state = useStore(socket.state, 'closed')
  return (
    <View style={state === 'open' ? styles.green : styles.grey} />
  )
}
```

Because `platform()` is installed above, `socket.online` is available too — the same `useStore` gives you an offline banner that follows NetInfo:

```tsx [OfflineBanner.tsx]
import { Banner } from './huddle-ui'
import { socket } from './socket'
import { useStore } from './use-store'

function OfflineBanner() {
  const online = useStore(socket.online, true)
  return online ? null : <Banner>You're offline</Banner>
}
```

Every other store binds through the exact same `useStore` — quality and latency from `keepalive()`, the reconnection loop, and a channel's roster and status:

```tsx [Indicators.tsx]
import { ActivityIndicator, FlatList, Text } from 'react-native'
import { Banner, Bars } from './huddle-ui'
import { socket } from './socket'
import { useStore } from './use-store'

const idle = { retrying: false, attempt: 0, nextRetryAt: null }

function SignalBars() {
  const quality = useStore(socket.quality, 'disconnected')
  const latency = useStore(socket.latency, null)
  return (
    <Bars
      tier={quality}
      title={latency === null ? 'measuring…' : `${latency}ms`}
    />
  )
}

function ReconnectBanner() {
  const { retrying, attempt } = useStore(socket.reconnection, idle)
  return retrying ? (
    <Banner>Reconnecting… (attempt {attempt})</Banner>
  ) : null
}

function Roster() {
  const room = socket.channel('rooms', 'room:general')
  const members = useStore(room.members, new Map())
  const status = useStore(room.status, 'unsubscribed')
  if (status === 'subscribing') return <ActivityIndicator />
  return (
    <FlatList
      data={[...members.values()]}
      renderItem={({ item }) => <Text>{String(item.meta.name)}</Text>}
    />
  )
}
```

## Receipts

Receipts are the same store shape — `dm.receivers` and `dm.readers` are each a `Map<userId, cursor>` (the `dms` family calls `inbox()` for receivers and `readReceipts()` for readers), so a per-message tick is one more `useStore`:

```tsx [Ticks.tsx]
import { Text } from 'react-native'
import { socket } from './socket'
import { useStore } from './use-store'

function Ticks({ pair, seq }: { pair: string; seq: number }) {
  const dm = socket.channel('dms', `dm:${pair}`)
  const receivers = useStore(dm.receivers)
  const readers = useStore(dm.readers)
  const read = [...readers.values()].some((m) => m >= seq)
  const received = [...receivers.values()].some((m) => m >= seq)
  return <Text>{read ? '✓✓ read' : received ? '✓✓' : '✓'}</Text>
}
```

## Surviving a cold start

If you use [durable streams](/client/streams/durable-and-continue), persist the `streamId` with `AsyncStorage` while the stream runs, and `continueStream(...)` with it after the app is killed and relaunched — the mobile equivalent of continuing after a page reload. The connection itself reconnects on its own through `reconnection()`; this is only for picking a long stream back up across a full restart.

```ts [useScoutStream.ts]
import AsyncStorage from '@react-native-async-storage/async-storage'
import { appendToken } from './huddle-ui'
import { socket } from './socket'

const KEY = 'scout-stream-id'

// While Scout's reply streams in, save its id so a relaunch can find it.
export async function startScoutStream(
  roomId: string,
  prompt: string,
) {
  const room = socket.channel('rooms', `room:${roomId}`)
  const handle = await room.stream('ask-scout', { prompt })
  await AsyncStorage.setItem(KEY, handle.streamId)
  for await (const chunk of handle) appendToken(chunk.text)
  await AsyncStorage.removeItem(KEY) // finished cleanly — nothing to continue
}

// On cold start, re-attach if we saved an id last time.
export async function continueScoutStreamIfAny(roomId: string) {
  const streamId = await AsyncStorage.getItem(KEY)
  if (!streamId) return
  const room = socket.channel('rooms', `room:${roomId}`)
  const continued = room.continueStream('ask-scout', streamId)
  for await (const chunk of continued) appendToken(chunk.text)
}
```
