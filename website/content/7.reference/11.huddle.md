Every example in these docs comes from one app: **Huddle**, a team chat app. People talk in rooms and direct messages, set a status, search the message history, and ask Scout, Huddle's AI assistant, questions. This page is the reference for its names and shapes. When an example uses an event, it has the payload shown here.

A page can show a smaller version of this server, with only the capabilities that page is about. The names and payloads stay the same.

## The server builder

`server/base.ts` holds the builder and the channels. Every handler and hook file imports from here.

```typescript [server/base.ts]
import { emits, errors, server, t } from '@livething/server'
import { channels } from '@livething/server/channels'
import { clientRequests } from '@livething/server/client-requests'
import { dedup } from '@livething/server/dedup'
import { CloseCodes, RejectUpgrade } from '@livething/server/errors'
import { inbox, readReceipts } from '@livething/server/inbox'
import { keepalive } from '@livething/server/keepalive'
import { replay } from '@livething/server/replay'
import { scheduled } from '@livething/server/scheduled'
import { sequence } from '@livething/server/sequence'
import { serverRequests } from '@livething/server/server-requests'
import { recordedStream, stream } from '@livething/server/streams'
import { tags } from '@livething/server/tags'
import * as v from 'valibot'
import { Forbidden, NotFound, RoomFull } from '../shared/errors'
import { verifySession } from './auth'
import { buildDigest } from './helpers'

export const Message = v.object({
  id: v.string(),
  text: v.string(),
  from: v.string(),
  sentAt: v.number(),
})
export type Message = v.InferOutput<typeof Message>

export const Status = v.picklist(['active', 'away', 'busy'])
export type Status = v.InferOutput<typeof Status>

export const base = server({
  async upgrade(request) {
    const token = new URL(request.url).searchParams.get('token')
    const user = await verifySession(token)
    if (!user)
      throw RejectUpgrade(CloseCodes.AUTH_INVALID, 'invalid session')
    return {
      userId: user.id,
      meta: { name: user.name, role: user.role },
    }
  },
  capabilities: [
    emits({
      notification: t<{ title: string; body: string }>(),
      announcement: t<{ text: string }>(),
      'user-renamed': t<{ userId: string; name: string }>(),
      'status-changed': t<{ userId: string; status: Status }>(),
      message: t<Message>(),
      reaction: t<{
        messageId: string
        emoji: string
        from: string
      }>(),
      'cursor-answer': t<{
        kind: 'received' | 'read'
        seq: number
        userIds: string[]
      }>(),
      health: t<{
        latencyMs: number | null
        quality: 'excellent' | 'good' | 'degraded' | 'poor' | null
      }>(),
    }),
    errors([NotFound, Forbidden, RoomFull]),
    channels(),
    keepalive(),
    scheduled({
      'send-digest': { data: t<{ userId: string }>() },
    }),
    tags((ephemeral) => [ephemeral('team', t<string>())]),
    clientRequests(),
    serverRequests({
      'confirm-delete': {
        out: t<{ preview: string }>(),
        in: v.object({ confirmed: v.boolean() }),
      },
    }),
    stream(),
    recordedStream(),
    dedup(),
    sequence({ persist: true }),
    replay.history(),
  ],
})

// A room: everyone in it sees its messages.
export const rooms = base.channel(['rooms', 'room:{roomId}'], {
  capabilities: [
    emits({
      message: t<Message>(),
      reaction: t<{
        messageId: string
        emoji: string
        from: string
      }>(),
      typing: t<{ from: string }>(),
    }),
  ],
})

// A direct message: every member gets a stored copy until they acknowledge it.
export const dms = base.channel(['dms', 'dm:{pair}'], {
  membership: 'persistent',
  capabilities: [
    inbox(),
    readReceipts(),
    emits({ message: t<Message>() }),
  ],
})

// Announcements: one fixed channel. Only the server sends to it.
export const announcements = base.channel('announcements', {
  capabilities: [emits({ announcement: t<{ text: string }>() })],
})

base.scheduled('send-digest', {
  run(ctx, data) {
    buildDigest(data.userId)
  },
})
```

## The definition

`server/huddle-definition.ts` adds the handlers and calls `assemble()`, which turns everything registered so far into the server definition. Anything added to `base` or a channel after `assemble()` runs is not part of it, so register hooks and scheduled tasks first.

```typescript [server/huddle-definition.ts]
import { t } from '@livething/server'
import * as v from 'valibot'
import type { InferClient } from '@livething/server'
import { auditLogEntries } from './audit-log'
import {
  announcements,
  base,
  dms,
  Message,
  rooms,
  Status,
} from './base'
import {
  deleteMessage,
  loadMessagePreview,
  loadMessages,
  loadProfile,
  runScoutStep,
  saveMessageReport,
  searchMessages,
} from './helpers'
import { llm } from './llm'

export const handleRename = base.on('rename', {
  input: v.object({ name: v.string() }),
  run(ctx, input) {
    ctx.broadcast('user-renamed', {
      userId: ctx.user.id,
      name: input.name,
    })
  },
})

export const handleSetStatus = base.on('set-status', {
  input: v.object({ status: Status }),
  run(ctx, input) {
    ctx.broadcast('status-changed', {
      userId: ctx.user.id,
      status: input.status,
    })
  },
})

export const handleGetProfile = base.request('get-profile', {
  input: v.object({ userId: v.string() }),
  run(ctx, input) {
    return loadProfile(input.userId)
  },
})

export const handleDeleteMessage = base.request('delete-message', {
  input: v.object({ messageId: v.string() }),
  async run(ctx, input) {
    const preview = await loadMessagePreview(input.messageId)
    const answer = await ctx.request('confirm-delete', { preview })
    if (!answer.confirmed) return { deleted: false }

    await deleteMessage(input.messageId)
    return { deleted: true }
  },
})

export const handleReportMessage = base.request('report-message', {
  input: v.object({ messageId: v.string(), reason: v.string() }),
  async run(ctx, input) {
    await saveMessageReport(
      input.messageId,
      input.reason,
      ctx.user.id,
    )
  },
})

export const handleSendMessage = rooms.on('send-message', {
  input: v.object({ text: v.string() }),
  run(ctx, input) {
    ctx.channel.broadcast('message', {
      id: crypto.randomUUID(),
      text: input.text,
      from: ctx.user.id,
      sentAt: Date.now(),
    })
  },
})

export const handleReact = rooms.on('react', {
  input: v.object({ messageId: v.string(), emoji: v.string() }),
  run(ctx, input) {
    ctx.channel.broadcast('reaction', {
      messageId: input.messageId,
      emoji: input.emoji,
      from: ctx.user.id,
    })
  },
})

export const handleStartTyping = rooms.on('start-typing', {
  input: v.object({}),
  run(ctx) {
    ctx.channel.broadcast('typing', { from: ctx.user.id })
  },
})

export const handleLoadHistory = rooms.request('load-history', {
  input: v.object({ before: v.optional(v.number()) }),
  async run(ctx, input) {
    return {
      messages: await loadMessages(
        ctx.channel.params.roomId,
        input.before,
      ),
    }
  },
})

export const handleSendDm = dms.on('send-message', {
  input: v.object({ text: v.string() }),
  run(ctx, input) {
    ctx.channel.broadcast('message', {
      id: crypto.randomUUID(),
      text: input.text,
      from: ctx.user.id,
      sentAt: Date.now(),
    })
  },
})

export const handleSearchMessages = base.stream('search-messages', {
  input: v.object({ query: v.string() }),
  chunk: t<{ hit: Message }>(),
  async *run(ctx, input) {
    for await (const hit of searchMessages(input.query)) yield { hit }
  },
})

export const handleRunTask = base.stream('run-task', {
  input: v.object({ goal: v.string(), roomId: v.string() }),
  chunk: t<{ step: string; detail: string }>(),
  nudges: {
    'add-context': v.object({ text: v.string() }),
  },
  async *run(ctx, input) {
    let context = ''
    for (const step of ['search', 'read', 'write'] as const) {
      const detail = await runScoutStep(step, input.goal, {
        context,
        signal: ctx.stream.signal,
      })
      const recent = ctx.stream.nudges(yield { step, detail })
      const added = recent.find((n) => n.event === 'add-context')
      if (added) context = added.payload.text
    }
  },
})

export const handleExportRoom = base.stream('export-room', {
  input: v.object({
    roomId: v.string(),
    before: v.optional(v.number()),
  }),
  chunk: t<{ messages: Message[] }>(),
  async *run(ctx, input) {
    let before = input.before
    while (true) {
      const messages = await loadMessages(input.roomId, before)
      if (messages.length === 0) return
      before = messages[messages.length - 1].sentAt
      yield { messages }
    }
  },
})

export const handleTailAuditLog = base.stream('tail-audit-log', {
  input: v.object({}),
  chunk: t<{ line: string }>(),
  nudges: {
    stop: null,
    'set-level': v.object({
      level: v.picklist(['info', 'warn', 'error']),
    }),
  },
  async *run(ctx) {
    const levels = ['info', 'warn', 'error'] as const
    let minLevel = 0
    for await (const entry of auditLogEntries(ctx.stream.signal)) {
      if (levels.indexOf(entry.level) < minLevel) continue

      const recent = ctx.stream.nudges(yield { line: entry.line })
      for (const nudge of recent) {
        if (nudge.event === 'stop') return
        if (nudge.event === 'set-level')
          minLevel = levels.indexOf(nudge.payload.level)
      }
    }
  },
})

export const handleAskScout = rooms.recordedStream('ask-scout', {
  input: v.object({ prompt: v.string() }),
  chunk: t<{ text: string }>(),
  nudges: {
    stop: null,
    redirect: v.object({ instruction: v.string() }),
  },
  broadcast: true, // every member of the room sees the reply
  async *run(ctx, input) {
    let messages = [{ role: 'user', text: input.prompt }]

    while (true) {
      const modelCall = new AbortController()
      let replySoFar = ''
      let redirectedTo: string | null = null

      for await (const piece of llm.stream({
        messages,
        signal: modelCall.signal,
      })) {
        replySoFar += piece
        const recent = ctx.stream.nudges(yield { text: piece })

        if (recent.some((n) => n.event === 'stop')) {
          modelCall.abort()
          return
        }

        const redirect = recent.find((n) => n.event === 'redirect')
        if (redirect) {
          modelCall.abort()
          redirectedTo = redirect.payload.instruction
          break
        }
      }

      if (!redirectedTo) return

      messages = [
        ...messages,
        { role: 'assistant', text: replySoFar },
        { role: 'user', text: redirectedTo },
      ]
    }
  },
})

export const huddle = base.assemble({
  handlers: [
    handleRename,
    handleSetStatus,
    handleGetProfile,
    handleDeleteMessage,
    handleReportMessage,
    handleSearchMessages,
    handleRunTask,
    handleExportRoom,
    handleTailAuditLog,
    rooms,
    handleAskScout,
    handleSendMessage,
    handleReact,
    handleStartTyping,
    handleLoadHistory,
    announcements,
    dms,
    handleSendDm,
  ],
  api: (ctx) => ({
    announce(text: string) {
      ctx.channels
        .get('announcements')
        .broadcast('announcement', { text })
    },
  }),
})

export type HuddleSocket = InferClient<typeof huddle>
```

## The client socket

```typescript [client/socket.ts]
import {
  channels,
  client,
  clientRequests,
  connect,
  dedup,
  gaps,
  inbox,
  keepalive,
  platform,
  readReceipts,
  reconnection,
  replay,
  serverRequests,
  stream,
} from '@livething/client'
import { WebEnvironment } from '@livething/client/environment'
import type { HuddleSocket } from '../server/huddle-definition'

export const socket = client<HuddleSocket>()(
  connect('wss://huddle.example.com/huddle'),
  channels(),
  clientRequests(),
  serverRequests(),
  stream(),
  inbox(),
  readReceipts(),
  dedup(),
  replay(),
  gaps(),
  keepalive(),
  platform({ environment: WebEnvironment }),
  reconnection(),
)
```
